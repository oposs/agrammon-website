{{- /* Link render hook. - /assets/… links are made base-URL aware (works under the GitHub Pages subpath): the leading slash is dropped before relURL, since relURL leaves root-relative paths untouched. - Links to PDFs get class "pdf" (shaped into download rows by partials/dl-rows.html). - External links get a ↗ marker. */ -}} {{- $u := .Destination -}} {{- $parsed := urls.Parse $u -}} {{- $external := in (slice "http" "https") $parsed.Scheme -}} {{- if strings.HasPrefix $u "/assets/" }}{{ $u = strings.TrimPrefix "/" $u | relURL }}{{ end -}} {{- if strings.HasSuffix (lower $parsed.Path) ".pdf" -}} {{ .Text | safeHTML }} {{- else if $external -}} {{ .Text | safeHTML }} ↗ {{- if eq .Page.Path "/links" }}{{ strings.TrimPrefix "www." $parsed.Host }}{{ end -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
