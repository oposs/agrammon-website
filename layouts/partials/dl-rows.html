{{- /* Turn PDF lists in rendered Markdown into download rows. Context: rendered HTML (.Content). Lists without any PDF link are left alone. In a PDF list, an item "Text — " (also --, - ) or consisting only of a PDF link becomes a row (text + "PDF ↓" button); any other item (no PDF yet, several PDF links) keeps its inline links. */ -}} {{- $html := string . -}} {{- $label := i18n "pdf_download" -}} {{- $withText := `
  • ([^<]*?(?:[^<]*?)*?)(?:\s*(?:—|—|–|–)\s*|\s+-\s+)]*>[^<]*\s*
    • ` -}} {{- $linkOnly := `
  • ]*>([^<]*)\s*
    • ` -}} {{- range findRE `(?s)
      .*?
    ` $html -}} {{- if strings.Contains . `class="pdf"` -}} {{- $list := replace . "
      " `
        ` 1 -}} {{- /* a blank line between bullets makes a loose list: unwrap

        */ -}} {{- $list = replaceRE `

      • \s*

        ` `

      • ` $list -}} {{- $list = replaceRE `

        \s*
        • ` `` $list -}} {{- $list = replaceRE $withText (printf `
      • $1%s
        • ` $label) $list -}} {{- $list = replaceRE $linkOnly (printf `
      • $2%s
        • ` $label) $list -}} {{- $html = replace $html . $list -}} {{- end -}} {{- end -}} {{- $html | safeHTML -}}
